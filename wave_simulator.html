5) create_system() button(text="Increase", bind=increase_points) button(text="Decrease", bind=decrease_points) # 🔥 Boundary Toggle scene.append_to_caption("\n\nRight End Condition:\n") def toggle_boundary(b): state["right_fixed"] = not state["right_fixed"] if state["right_fixed"]: b.text = "Fixed End" else: b.text = "Free End" boundary_button = button(text="Fixed End", bind=toggle_boundary) # --- Animation --- t = 0 dt = 0.02 while True: rate(200) omega = 2 * pi * state["freq"] k = 2 * pi * state["freq"] / v A = state["amp"] string_curve.clear() string_curve.append(vector(0, 0, 0)) for p in particles: x = p.pos.x if state["right_fixed"]: # 🔵 Fixed-Fixed → standing wave y = A * sin(k*x - intensity) string_curve.append(p.pos) # Right endpoint behavior if state["right_fixed"]: string_curve.append(vector(L, 0, 0)) else: # Free end moves → take last particle height string_curve.append(vector(L, particles[-1].pos.y, 0)) t += dt
