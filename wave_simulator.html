from vpython import * # --- Scene --- scene = canvas(title="Melde's Experiment Simulation", width=1200, height=600, background=color.black) L = 20 scene.center = vector(L/2, 0, 0) scene.range = L/2 + 5 scene.autoscale = False # --- State --- state = { "freq": 0.5, "amp": 1.5, "points": 80, "damping": 0.01, "right_fixed": True } particles = [] string_curve = None # Wave arrays y = [] y_prev = [] y_next = [] v = 10 # wave speed dt = 0.005 # --- Create system --- def create_system(): global particles, string_curve, y, y_prev, y_next for p in particles: p.visible = False if string_curve: string_curve.visible = False particles = [] n = state["points"] dx = L / (n + 1) y = [0]*(n+2) y_prev = [0]*(n+2) y_next = [0]*(n+2) string_curve = curve(color=color.white, radius=0.05) for i in range(n+2): x = i*dx if i == 0 or i == n+1: string_curve.append(vector(x, 0, 0)) else: p = sphere(pos=vector(x, 0, 0), radius=0.22, color=color.cyan) particles.append(p) string_curve.append(p.pos) create_system() # Walls box(pos=vector(0,0,0), size=vector(0.2,4,0.2), color=color.gray(0.7)) box(pos=vector(L,0,0), size=vector(0.2,4,0.2), color=color.gray(0.7)) # --- UI --- scene.append_to_caption("\n") def update_freq(s): state["freq"] = s.value freq_label.text = f"{state['freq']:.2f}" scene.append_to_caption("Freq ") slider(min=0.1, max=1.5, value=0.5, step=0.01, bind=update_freq) freq_label = wtext(text="0.50 ") def update_amp(s): state["amp"] = s.value amp_label.text = f"{state['amp']:.2f}" scene.append_to_caption(" Amp ") slider(min=0.5, max=3, value=1.5, step=0.1, bind=update_amp) amp_label = wtext(text="1.50 ") def update_damping(s): state["damping"] = s.value damp_label.text = f"{state['damping']:.3f}" scene.append_to_caption(" Damp ") slider(min=0, max=0.1, value=0.01, step=0.001, bind=update_damping) damp_label = wtext(text="0.01 ") scene.append_to_caption(" Particles ") def set_particles(inp): try: val = int(inp.text) if val >= 1: state["points"] = val create_system() except: pass winput(bind=set_particles, text="80") scene.append_to_caption(" ") def toggle_boundary(b): state["right_fixed"] = not state["right_fixed"] b.text = "Fixed" if state["right_fixed"] else "Free" button(text="Fixed", bind=toggle_boundary) # --- Simulation loop --- t = 0 while True: rate(200) n = state["points"] dx = L/(n+1) c = v * dt / dx # Stability condition if c > 1: continue # --- Left end (driven) --- y[0] = state["amp"] * sin(2*pi*state["freq"]*t) # --- Update interior --- for i in range(1, n+1): y_next[i] = (2*y[i] - y_prev[i] + c**2*(y[i+1] - 2*y[i] + y[i-1]) - state["damping"]*(y[i] - y_prev[i])) # --- Right boundary --- if state["right_fixed"]: y_next[n+1] = 0 # node else: y_next[n+1] = y_next[n] # free end (zero slope) # --- Update arrays --- y_prev = y[:] y = y_next[:] # --- Update visuals --- for i, p in enumerate(particles): p.pos.y = y[i+1] intensity = abs(y[i+1]) / (state["amp"] + 1e-5) p.color = vector(intensity, 0.8, 1-intensity) string_curve.modify(i+1, p.pos) t += dt
